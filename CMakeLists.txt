cmake_minimum_required(VERSION 3.16)
project(Beryl VERSION 0.5.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Beryl requires Linux (on Windows, build inside WSL).")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BERYL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_compile_definitions(_POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes)
if(BERYL_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(beryl_core src/files.c src/hashes.c src/workspace.c src/runner.c src/oracle.c src/results.c
  src/copybook.c src/record_codec.c src/generator.c src/report.c src/options.c src/mutator.c src/campaign.c src/minimizer.c src/driver.c src/compiler_gnucobol.c src/coverage.c src/fixtures.c)
target_include_directories(beryl_core PUBLIC include)
add_executable(beryl src/main.c src/cli.c)
target_link_libraries(beryl PRIVATE beryl_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/c/test_core.c)
  target_link_libraries(test_core PRIVATE beryl_core)
  add_executable(target_fixture tests/c/target_fixture.c)
  # The deliberately crashing child must retain normal signal semantics.
  if(BERYL_SANITIZERS)
    target_compile_options(target_fixture PRIVATE -fno-sanitize=all)
    target_link_options(target_fixture PRIVATE -fno-sanitize=all)
  endif()
  add_executable(test_acceptance tests/c/test_acceptance.c)
  target_link_libraries(test_acceptance PRIVATE beryl_core)
  target_compile_options(test_core PRIVATE -UNDEBUG)
  target_compile_options(test_acceptance PRIVATE -UNDEBUG)
  add_executable(test_records tests/c/test_records.c)
  target_link_libraries(test_records PRIVATE beryl_core)
  target_compile_options(test_records PRIVATE -UNDEBUG)
  add_test(NAME records COMMAND test_records)
  add_executable(test_generate tests/c/test_generate.c)
  target_link_libraries(test_generate PRIVATE beryl_core)
  target_compile_options(test_generate PRIVATE -UNDEBUG)
  add_test(NAME generation COMMAND test_generate $<TARGET_FILE:beryl>)
  add_executable(test_reports tests/c/test_reports.c)
  target_link_libraries(test_reports PRIVATE beryl_core)
  target_compile_options(test_reports PRIVATE -UNDEBUG)
  add_test(NAME reports COMMAND test_reports)
  set_tests_properties(records generation reports PROPERTIES TIMEOUT 60)
  add_test(NAME core COMMAND test_core)
  add_test(NAME acceptance COMMAND test_acceptance $<TARGET_FILE:beryl> $<TARGET_FILE:target_fixture>)
  set_tests_properties(acceptance PROPERTIES TIMEOUT 60)
  add_executable(reference_fixture tests/c/differential_fixture.c)
  add_executable(candidate_fixture tests/c/differential_fixture.c)
  target_compile_definitions(candidate_fixture PRIVATE DIFF_BUG=1)
  if(BERYL_SANITIZERS)
    foreach(fixture reference_fixture candidate_fixture)
      target_compile_options(${fixture} PRIVATE -fno-sanitize=all)
      target_link_options(${fixture} PRIVATE -fno-sanitize=all)
    endforeach()
  endif()
  add_executable(test_reference tests/c/test_reference.c)
  target_link_libraries(test_reference PRIVATE beryl_core)
  target_compile_options(test_reference PRIVATE -UNDEBUG)
  add_test(NAME reference COMMAND test_reference $<TARGET_FILE:beryl> $<TARGET_FILE:reference_fixture> $<TARGET_FILE:candidate_fixture>)
  set_tests_properties(reference PROPERTIES TIMEOUT 60)
  add_executable(test_campaign tests/c/test_campaign.c)
  target_link_libraries(test_campaign PRIVATE beryl_core)
  target_compile_options(test_campaign PRIVATE -UNDEBUG)
  add_test(NAME campaign COMMAND test_campaign $<TARGET_FILE:beryl> $<TARGET_FILE:reference_fixture> $<TARGET_FILE:candidate_fixture>)
  set_tests_properties(campaign PROPERTIES TIMEOUT 90)
  add_executable(test_minimizer tests/c/test_minimizer.c)
  target_link_libraries(test_minimizer PRIVATE beryl_core)
  target_compile_options(test_minimizer PRIVATE -UNDEBUG)
  add_test(NAME minimizer COMMAND test_minimizer $<TARGET_FILE:beryl> $<TARGET_FILE:target_fixture>)
  set_tests_properties(minimizer PROPERTIES TIMEOUT 90)
  add_executable(multifile_fixture tests/c/multifile_fixture.c)
  if(BERYL_SANITIZERS)
    target_compile_options(multifile_fixture PRIVATE -fno-sanitize=all)
    target_link_options(multifile_fixture PRIVATE -fno-sanitize=all)
  endif()
  add_executable(test_fixtures tests/c/test_fixtures.c)
  target_link_libraries(test_fixtures PRIVATE beryl_core)
  target_compile_options(test_fixtures PRIVATE -UNDEBUG)
  add_test(NAME fixtures COMMAND test_fixtures $<TARGET_FILE:beryl> $<TARGET_FILE:multifile_fixture>)
  set_tests_properties(fixtures PROPERTIES TIMEOUT 90)
  add_executable(test_driver tests/c/test_driver.c)
  target_link_libraries(test_driver PRIVATE beryl_core)
  target_compile_options(test_driver PRIVATE -UNDEBUG)
  add_test(NAME driver COMMAND test_driver $<TARGET_FILE:beryl>)
  add_executable(test_formats tests/c/test_formats.c)
  target_link_libraries(test_formats PRIVATE beryl_core)
  target_compile_options(test_formats PRIVATE -UNDEBUG)
  add_test(NAME formats COMMAND test_formats $<TARGET_FILE:beryl>)
  add_executable(test_coverage tests/c/test_coverage.c)
  target_link_libraries(test_coverage PRIVATE beryl_core)
  target_compile_options(test_coverage PRIVATE -UNDEBUG)
  find_program(COBC_EXECUTABLE cobc)
  if(COBC_EXECUTABLE)
    add_test(NAME cobol_coverage COMMAND test_coverage $<TARGET_FILE:beryl> ${COBC_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cobol/accounts.cob)
    set_tests_properties(cobol_coverage PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
    add_test(NAME cobol_preprocessing COMMAND test_formats $<TARGET_FILE:beryl> ${COBC_EXECUTABLE})
    set_tests_properties(cobol_preprocessing PROPERTIES TIMEOUT 60)
    add_test(NAME cobol_driver COMMAND test_driver $<TARGET_FILE:beryl> ${COBC_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cobol/clamp_subprogram.cob)
    set_tests_properties(cobol_driver PROPERTIES TIMEOUT 30)
    foreach(fixture accounts accounts_bug codec_vectors generated_records)
      add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${fixture}
        COMMAND ${COBC_EXECUTABLE} -x -free -o ${CMAKE_CURRENT_BINARY_DIR}/${fixture} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cobol/${fixture}.cob
        DEPENDS tests/cobol/${fixture}.cob VERBATIM)
      add_custom_target(${fixture}_fixture ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/${fixture})
    endforeach()
    add_test(NAME cobol_acceptance COMMAND test_acceptance $<TARGET_FILE:beryl> ${CMAKE_CURRENT_BINARY_DIR}/accounts ${CMAKE_CURRENT_BINARY_DIR}/accounts_bug ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(cobol_acceptance PROPERTIES TIMEOUT 30)
    add_test(NAME cobol_codec COMMAND test_records --cobol $<TARGET_FILE:beryl> ${CMAKE_CURRENT_BINARY_DIR}/codec_vectors)
    set_tests_properties(cobol_codec PROPERTIES TIMEOUT 30)
    add_test(NAME cobol_generation COMMAND test_generate $<TARGET_FILE:beryl> ${CMAKE_CURRENT_BINARY_DIR}/generated_records)
    set_tests_properties(cobol_generation PROPERTIES TIMEOUT 60)
  else()
    add_test(NAME cobol_coverage COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_coverage PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cobol_preprocessing COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_preprocessing PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cobol_driver COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_driver PROPERTIES SKIP_RETURN_CODE 77)
    message(STATUS "SKIP: GnuCOBOL not found; C-only tests remain enabled")
    add_test(NAME cobol_acceptance COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_acceptance PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cobol_codec COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_codec PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cobol_generation COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_generation PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  include(cmake/CompilerMatrix.cmake)
endif()
install(TARGETS beryl RUNTIME DESTINATION bin)
