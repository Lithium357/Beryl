cmake_minimum_required(VERSION 3.16)
project(Beryl VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Beryl Phase 1 requires Linux (on Windows, build inside WSL).")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BERYL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_compile_definitions(_POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes)
if(BERYL_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(beryl_core src/files.c src/hashes.c src/workspace.c src/runner.c src/oracle.c src/results.c)
target_include_directories(beryl_core PUBLIC include)
add_executable(beryl src/main.c src/cli.c)
target_link_libraries(beryl PRIVATE beryl_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/c/test_core.c)
  target_link_libraries(test_core PRIVATE beryl_core)
  add_executable(target_fixture tests/c/target_fixture.c)
  # The deliberately crashing child must retain normal signal semantics.
  if(BERYL_SANITIZERS)
    target_compile_options(target_fixture PRIVATE -fno-sanitize=all)
    target_link_options(target_fixture PRIVATE -fno-sanitize=all)
  endif()
  add_executable(test_acceptance tests/c/test_acceptance.c)
  target_link_libraries(test_acceptance PRIVATE beryl_core)
  target_compile_options(test_core PRIVATE -UNDEBUG)
  target_compile_options(test_acceptance PRIVATE -UNDEBUG)
  add_test(NAME core COMMAND test_core)
  add_test(NAME acceptance COMMAND test_acceptance $<TARGET_FILE:beryl> $<TARGET_FILE:target_fixture>)
  set_tests_properties(acceptance PROPERTIES TIMEOUT 60)
  find_program(COBC_EXECUTABLE cobc)
  if(COBC_EXECUTABLE)
    foreach(fixture accounts accounts_bug)
      add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${fixture}
        COMMAND ${COBC_EXECUTABLE} -x -free -o ${CMAKE_CURRENT_BINARY_DIR}/${fixture} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cobol/${fixture}.cob
        DEPENDS tests/cobol/${fixture}.cob VERBATIM)
      add_custom_target(${fixture}_fixture ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/${fixture})
    endforeach()
    add_test(NAME cobol_acceptance COMMAND test_acceptance $<TARGET_FILE:beryl> ${CMAKE_CURRENT_BINARY_DIR}/accounts ${CMAKE_CURRENT_BINARY_DIR}/accounts_bug ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(cobol_acceptance PROPERTIES TIMEOUT 30)
  else()
    message(STATUS "SKIP: GnuCOBOL not found; C-only tests remain enabled")
    add_test(NAME cobol_acceptance COMMAND test_acceptance --skip-cobol)
    set_tests_properties(cobol_acceptance PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
install(TARGETS beryl RUNTIME DESTINATION bin)
